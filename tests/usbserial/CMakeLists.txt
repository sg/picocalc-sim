cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(usbserial C CXX ASM)
pico_sdk_init()
add_executable(usbserial usbserial.c)
target_link_libraries(usbserial pico_stdlib)
pico_enable_stdio_usb(usbserial 1)
pico_enable_stdio_uart(usbserial 0)
pico_add_extra_outputs(usbserial)
