cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(arduino C CXX ASM)
pico_sdk_init()
add_executable(arduino arduino.c)
target_link_libraries(arduino pico_stdlib hardware_i2c hardware_pio)
pico_enable_stdio_usb(arduino 0)
pico_enable_stdio_uart(arduino 1)
pico_add_extra_outputs(arduino)
